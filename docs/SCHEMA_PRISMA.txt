model Tenant {
  id        String   @id @default(cuid())
  name      String
  slug      String   @unique
  isActive  Boolean  @default(true)
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  users     User[]
  doctors   Doctor[]
  patients  Patient[]
  pharmacies Pharmacy[]
  wallets   Wallet[]
  auditLogs AuditLog[]
  consents  Consent[]
  @@index([isActive])
}

model User {
  id            String     @id @default(cuid())
  tenantId      String?
  tenant        Tenant?    @relation(fields: [tenantId], references: [id], onDelete: SetNull)

  email         String     @unique
  passwordHash  String
  role          UserRole
  status        UserStatus @default(ACTIVE)
  fullName      String
  phone         String?
  lastLoginAt   DateTime?

  // If you want 1:1 mapping:
  doctor        Doctor?
  patient       Patient?
  pharmacyUser  PharmacyUser?

  sessions      Session[]
  mfaSecrets    MfaSecret[]
  auditLogs     AuditLog[] @relation("AuditActor")

  createdAt     DateTime   @default(now())
  updatedAt     DateTime   @updatedAt

  @@index([tenantId, role])
  @@index([status])
}

model Session {
  id             String   @id @default(cuid())
  userId         String
  user           User     @relation(fields: [userId], references: [id], onDelete: Cascade)

  refreshTokenHash String
  accessIssuedAt DateTime @default(now())
  refreshExpiresAt DateTime

  deviceHash     String?
  ipMasked       String?
  userAgentHash  String?

  revokedAt      DateTime?
  createdAt      DateTime @default(now())

  @@index([userId])
  @@index([refreshExpiresAt])
}

model MfaSecret {
  id         String   @id @default(cuid())
  userId     String
  user       User     @relation(fields: [userId], references: [id], onDelete: Cascade)

  type       MfaType
  secretEnc  String   // encrypted secret
  enabledAt  DateTime?
  verifiedAt DateTime?
  createdAt  DateTime @default(now())

  @@index([userId, type])
}

model Doctor {

model AppointmentSlot {
  id         String   @id @default(cuid())
  doctorId   String
  doctor     Doctor   @relation(fields: [doctorId], references: [id], onDelete: Cascade)

  startsAt   DateTime
  endsAt     DateTime

  createdAt  DateTime @default(now())

  @@unique([doctorId, startsAt, endsAt])
  @@index([doctorId, startsAt])
}

model Appointment {
  id           String            @id @default(cuid())
  tenantId     String?
  tenant       Tenant?           @relation(fields: [tenantId], references: [id], onDelete: SetNull)

  status       AppointmentStatus @default(DRAFT)

  doctorId     String
  doctor       Doctor            @relation(fields: [doctorId], references: [id], onDelete: Restrict)
  patientId    String
  patient      Patient           @relation(fields: [patientId], references: [id], onDelete: Restrict)

  slotId       String?           @unique
  slot         AppointmentSlot?  @relation(fields: [slotId], references: [id], onDelete: SetNull)

  // Pay-first-schedule-after
  paymentIntentId String?
  paymentIntent   PaymentIntent? @relation(fields: [paymentIntentId], references: [id], onDelete: SetNull)

  // Telehealth
  videoSession    VideoSession?

  startedAt   DateTime?
  endedAt     DateTime?

  notes       String?

  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  @@index([tenantId])
  @@index([doctorId, status])
  @@index([patientId, status])
}

model MedicalRecord {

model RecordEntry {
  id            String      @id @default(cuid())
  medicalRecordId String
  medicalRecord  MedicalRecord @relation(fields: [medicalRecordId], references: [id], onDelete: Cascade)

  authorUserId   String
  authorUser     User        @relation(fields: [authorUserId], references: [id], onDelete: Restrict)

model Document {
  id            String      @id @default(cuid())
  medicalRecordId String?
  medicalRecord  MedicalRecord? @relation(fields: [medicalRecordId], references: [id], onDelete: SetNull)

  appointmentId  String?
  appointment    Appointment? @relation(fields: [appointmentId], references: [id], onDelete: SetNull)

  prescriptionId String?
  prescription   Prescription? @relation(fields: [prescriptionId], references: [id], onDelete: SetNull)

  type           String      // PRESCRIPTION|LAUDO|ATESTADO|TCLE etc.
  fileUrl        String      // stored in S3; access via signed URLs
  hashPre        String?
  hashPost       String?
  signedByUserId String?
  signedByUser   User?       @relation(fields: [signedByUserId], references: [id], onDelete: SetNull)
  signedAt       DateTime?

  createdAt      DateTime    @default(now())

  @@index([type])
  @@index([appointmentId])
  @@index([prescriptionId])
}

model Prescription {
  id           String   @id @default(cuid())
  appointmentId String
  appointment  Appointment @relation(fields: [appointmentId], references: [id], onDelete: Cascade)

  doctorId     String
  doctor       Doctor   @relation(fields: [doctorId], references: [id], onDelete: Restrict)

  patientId    String
  patient      Patient  @relation(fields: [patientId], references: [id], onDelete: Restrict)

  status       String   @default("DRAFT") // DRAFT|SIGNED|SNCR_REGISTERED|CANCELLED
  productName  String
  thcPercent   Float?
  cbdPercent   Float?
  dosage       String
  durationDays Int?
  quantityText String?

  sncrRequired Boolean @default(false)
  sncrNumber   String?
  sncrStatus   String?

  documents    Document[]

  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  @@index([doctorId, createdAt])
  @@index([patientId, createdAt])
  @@index([sncrRequired, sncrStatus])
}

model Wallet {
  id        String          @id @default(cuid())
  tenantId  String?
  tenant    Tenant?         @relation(fields: [tenantId], references: [id], onDelete: SetNull)

  ownerType WalletOwnerType
  ownerId   String?         // doctorId/pharmacyId or null for platform/gateway
  currency  String          @default("BRL")

  balances  WalletBalance?
  entries   LedgerEntry[]

  createdAt DateTime @default(now())

  @@index([tenantId])
  @@index([ownerType, ownerId])
}

model WalletBalance {
  id              String  @id @default(cuid())
  walletId        String  @unique
  wallet          Wallet  @relation(fields: [walletId], references: [id], onDelete: Cascade)

  balanceTotal    Int     @default(0)
  balancePending  Int     @default(0)
  balanceAvailable Int    @default(0)
  balanceOnHold   Int     @default(0)
  balanceWithdrawn Int    @default(0)

  updatedAt       DateTime @updatedAt
}

model LedgerEntry {
  id            String          @id @default(cuid())
  tenantId      String?
  tenant        Tenant?         @relation(fields: [tenantId], references: [id], onDelete: SetNull)

  walletId      String
  wallet        Wallet          @relation(fields: [walletId], references: [id], onDelete: Cascade)

  direction     LedgerDirection
  amountCents   Int
  currency      String          @default("BRL")
  entryType     LedgerEntryType

  referenceType ResourceType
  referenceId   String
  idempotencyKey String?

  createdAt     DateTime @default(now())

  @@index([walletId, createdAt])
  @@index([referenceType, referenceId])
  @@index([tenantId])
  @@unique([walletId, idempotencyKey])
}
